package com.example.demo;

import com.example.demo.repository.voicetotextrepository; // Import your repository
import com.example.demo.service.AudioRecorder;
import com.example.demo.service.VoiceRecognitionService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.CommandLineRunner; // Use CommandLineRunner for startup logic
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.context.annotation.Bean;

import java.io.FileWriter;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

@SpringBootApplication
public class DemoApplication {

    public static void main(String[] args) {
        SpringApplication.run(DemoApplication.class, args);
    }

    @Bean
    public CommandLineRunner run(@Autowired voicetotextrepository repository) {
        return args -> {
            AudioRecorder recorder = new AudioRecorder();
            VoiceRecognitionService voiceService = new VoiceRecognitionService(repository);
            Scanner scanner = new Scanner(System.in);

            List<String> currentSessionValues = new ArrayList<>(); // List to hold current session values

            try {
                // Define a keyword list with phonetic approximations for complex words
                String keywordList = "[\"hello\",[\"hello world\", \"static\", \"print\", \"void\", \"main\", \"semi colon\", \"hi\", " +
                        "\"parenthesis\", \"puh-ren-thuh-sis\", \"pra-va-li-ka\", " +
                        "\"open parenthesis\", \"ˈoʊpən pəˈrɛnθɪsɪs\", \"addition\",\"d q\", \"ah-dih-shun\", \"comma\", \"plus\", \"plʌs\"]";

                // Load the Vosk model once with the custom keyword list
                voiceService.loadModel("src/main/resources/vosk-model-en-in-0.5", keywordList);

                // Infinite loop to keep asking for new input
                while (true) {
                    // Prompt user to start speaking
                    System.out.println("Press 'Enter' to start speaking...");
                    scanner.nextLine();  // Wait for the user to press enter

                    // Record audio for a fixed duration (e.g., 5 seconds)
                    String audioFilePath = "output.wav"; // Local path for the output audio file
                    recorder.recordAudio(audioFilePath, 5); // Adjust duration as needed

                    // Recognize from the recorded audio file
                    voiceService.recognizeFromFile(audioFilePath); // Ensure this matches your recorded file path

                    // Add recognized values to the current session
                    currentSessionValues.addAll(voiceService.getRecognizedValues()); // Add recognized values to the current session

                    // Check for exit command
                    if (currentSessionValues.contains("please exit")) {
                        System.out.println("Please enter the file name to save the recognized values: ");
                        String fileName = scanner.nextLine(); // Ask for the file name

                        // Write recognized values of the current session to the specified .txt file
                        writeValuesToFile(currentSessionValues, fileName); // Write all current session values

                        openNotepad(fileName); // Open the file in Notepad

                        currentSessionValues.clear(); // Clear current session values for the next session
                        continue; // Continue to ask for new input
                    }

                    // Ask user if they want to continue or exit
                    System.out.println("Do you want to record another audio? (yes/no)");
                    String response = scanner.nextLine().trim().toLowerCase();
                    if (response.equals("no")) {
                        // Ask for the file name before saving and opening it in Notepad
                        System.out.println("Please enter the file name to save the recognized values: ");
                        String fileName = scanner.nextLine(); // Ask for the file name

                        // Write recognized values of the current session to the specified .txt file
                        writeValuesToFile(currentSessionValues, fileName);

                        // Open the file in Notepad
                        openNotepad(fileName);

                        break; // Exit the loop
                    }
                }

            } catch (IOException e) {
                e.printStackTrace();
            } finally {
                scanner.close();  // Close the scanner resource
            }
        };
    }

    // Method to write recognized values to a .txt file
    private void writeValuesToFile(List<String> recognizedValues, String fileName) throws IOException {
        try (FileWriter fileWriter = new FileWriter(fileName, true)) { // Append mode
            String line = String.join(" ", recognizedValues); // Join all recognized values with a space
            fileWriter.write(line + System.lineSeparator()); // Write to file followed by a newline
        }
        System.out.println("Recognized values saved to " + fileName);
    }

    // Method to open the .txt file in Notepad
    private void openNotepad(String filePath) {
        try {
            Runtime.getRuntime().exec("notepad " + filePath);
            System.out.println("Opening " + filePath + " in Notepad...");
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}

package com.example.demo.service;

import com.example.demo.model.programodel;
import com.example.demo.repository.voicetotextrepository;
import edu.stanford.nlp.ling.CoreAnnotations;
import edu.stanford.nlp.ling.CoreLabel;
import edu.stanford.nlp.pipeline.*;
import org.vosk.Model;
import org.vosk.Recognizer;

import javax.sound.sampled.*;
import java.io.File;
import java.io.FileWriter;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.Optional;
import java.util.Properties;
import java.util.Scanner;

public class VoiceRecognitionService {
    private Recognizer recognizer;
    private final voicetotextrepository repository;
    private final List<String> recognizedValues; // List to store recognized values
    private StanfordCoreNLP stanfordNLP;  // NLP pipeline
    private final List<String> sessionRecognizedValues; // List to store recognized values for the session
    private Scanner scanner; // Scanner for user input

    public VoiceRecognitionService(voicetotextrepository repository) {
        this.repository = repository;
        this.recognizedValues = new ArrayList<>(); // Initialize the list
        this.sessionRecognizedValues = new ArrayList<>(); // Initialize session list
        this.scanner = new Scanner(System.in); // Initialize scanner
        initStanfordNLP();  // Initialize NLP on service creation
    }

    // Initialize Stanford CoreNLP pipeline
    private void initStanfordNLP() {
        Properties props = new Properties();
        props.setProperty("annotators", "tokenize,ssplit,pos,lemma,ner");  // Annotators for tokenization, POS, NER, etc.
        stanfordNLP = new StanfordCoreNLP(props);
    }

    // Load the Vosk model from the specified path with custom keyword list
    public void loadModel(String modelPath, String keywordList) throws IOException {
        Model model = new Model(modelPath);
        recognizer = new Recognizer(model, 16000.0f, keywordList);
    }

    // Recognize speech from an audio file and process with NLP
    public void recognizeFromFile(String filePath) throws IOException {
        try {
            File audioFile = new File(filePath);
            AudioInputStream audioInputStream = AudioSystem.getAudioInputStream(audioFile);

            byte[] buffer = new byte[4096];
            int bytesRead;

            while ((bytesRead = audioInputStream.read(buffer)) != -1) {
                recognizer.acceptWaveForm(buffer, bytesRead);
            }

            String jsonResult = recognizer.getFinalResult();
            System.out.println("Final Recognition Result: " + jsonResult);

            String recognizedText = extractRecognizedText(jsonResult);
            System.out.println("Recognized Text: '" + recognizedText + "'");

            // Process the recognized text with NLP
            processTextWithNLP(recognizedText);

            // Check for the exit command
            if (recognizedText.equalsIgnoreCase("please exit")) {
                // Print the recognized values of the current session to the console
                System.out.println("Recognized values for this session: " + String.join(" ", sessionRecognizedValues));

                // Ask the user for the file name
                System.out.println("Please enter the file name to save the recognized values: ");
                String fileName = scanner.nextLine(); // Ask for the file name

                // Append recognized values of the current session to the specified .txt file
                writeValuesToFile(sessionRecognizedValues, fileName);

                // Open the file in Notepad (optional)
                openNotepad(fileName);

                sessionRecognizedValues.clear(); // Clear session values after writing
                System.out.println("Session ended. Continuing to record...");
                return; // Exit the current method
            } else {
                // Fetch value from the database based on recognized text
                Optional<programodel> commandValue = repository.findByCommand(recognizedText.toLowerCase().trim());
                if (commandValue.isPresent()) {
                    // Add the recognized value to the session list
                    sessionRecognizedValues.add(commandValue.get().getValue());
                    recognizedValues.add(commandValue.get().getValue()); // Add to all recognized values
                    System.out.println("Recognized Value: " + commandValue.get().getValue());
                } else {
                    System.out.println("Command not found in database.");
                }
            }

            audioInputStream.close();

        } catch (UnsupportedAudioFileException e) {
            System.err.println("Unsupported audio format. Please ensure the audio is in 16kHz, 16-bit PCM WAV format.");
            e.printStackTrace();
        }
    }

    // Process the recognized text with NLP
    private void processTextWithNLP(String text) {
        // Create a CoreDocument with the text
        CoreDocument document = new CoreDocument(text);

        // Annotate the document with the NLP pipeline
        stanfordNLP.annotate(document);

        // Print tokens, part-of-speech tags, and named entities for analysis
        for (CoreLabel token : document.tokens()) {
            String word = token.word();
            String pos = token.get(CoreAnnotations.PartOfSpeechAnnotation.class);
            String ner = token.get(CoreAnnotations.NamedEntityTagAnnotation.class);

            System.out.println("Token: " + word + ", POS: " + pos + ", NER: " + ner);
        }
    }

    // Extract recognized text from JSON result
    private String extractRecognizedText(String jsonResult) {
        return jsonResult.replaceAll("^\\{\\s*\"text\"\\s*:\\s*\"([^\"]+)\"\\s*\\}$", "$1");
    }

    // Method to append recognized values to a .txt file
    private void writeValuesToFile(List<String> values, String fileName) {
        try (FileWriter fileWriter = new FileWriter(fileName, true)) { // Append mode
            for (String value : values) {
                fileWriter.write(value + " "); // Write each value followed by a space
            }
            fileWriter.write(System.lineSeparator()); // Add a newline after all values
            System.out.println("Appended recognized values to " + fileName); // Debug print
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    // Method to open a file in Notepad (optional)
    private void openNotepad(String fileName) {
        try {
            Runtime.getRuntime().exec("notepad.exe " + fileName); // Open the specified file in Notepad
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    // Getter for recognized values
    public List<String> getRecognizedValues() {
        return recognizedValues; // Return all recognized values
    }

    // Getter for session recognized values
    public List<String> getSessionRecognizedValues() {
        return sessionRecognizedValues; // Return the current session's recognized values
    }
}

package com.example.demo.service;

import javax.sound.sampled.*;
import java.io.File;
import java.io.IOException;

public class AudioRecorder {
    private AudioFormat audioFormat;
    private TargetDataLine targetLine;

    public void recordAudio(String filePath, int durationInSeconds) {
        audioFormat = new AudioFormat(16000, 16, 1, true, true); // PCM format
        DataLine.Info info = new DataLine.Info(TargetDataLine.class, audioFormat);

        try {
            targetLine = (TargetDataLine) AudioSystem.getLine(info);
            targetLine.open(audioFormat);
            targetLine.start();

            System.out.println("Recording...");

            AudioInputStream audioInputStream = new AudioInputStream(targetLine);
            File audioFile = new File(filePath);
            // Start recording
            new Thread(() -> {
                try {
                    Thread.sleep(durationInSeconds * 1000);
                } catch (InterruptedException e) {
                    e.printStackTrace();
                }
                targetLine.stop();
                targetLine.close();
                System.out.println("Recording finished.");
            }).start();

            // Write audio data to file
            AudioSystem.write(audioInputStream, AudioFileFormat.Type.WAVE, audioFile);
        } catch (LineUnavailableException e) {
            e.printStackTrace();
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}

package com.example.demo.service;

import com.example.demo.model.programodel;
import com.example.demo.repository.voicetotextrepository;
import edu.stanford.nlp.ling.CoreAnnotations;
import edu.stanford.nlp.ling.CoreLabel;
import edu.stanford.nlp.pipeline.*;
import org.vosk.Model;
import org.vosk.Recognizer;

import javax.sound.sampled.*;
import java.io.File;
import java.io.FileWriter;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.Optional;
import java.util.Properties;
import java.util.Scanner;

public class VoiceRecognitionService {
    private Recognizer recognizer;
    private final voicetotextrepository repository;
    private final List<String> recognizedValues; // List to store recognized values
    private StanfordCoreNLP stanfordNLP;  // NLP pipeline
    private final List<String> sessionRecognizedValues; // List to store recognized values for the session
    private Scanner scanner; // Scanner for user input

    public VoiceRecognitionService(voicetotextrepository repository) {
        this.repository = repository;
        this.recognizedValues = new ArrayList<>(); // Initialize the list
        this.sessionRecognizedValues = new ArrayList<>(); // Initialize session list
        this.scanner = new Scanner(System.in); // Initialize scanner
        initStanfordNLP();  // Initialize NLP on service creation
    }

    // Initialize Stanford CoreNLP pipeline
    private void initStanfordNLP() {
        Properties props = new Properties();
        props.setProperty("annotators", "tokenize,ssplit,pos,lemma,ner");  // Annotators for tokenization, POS, NER, etc.
        stanfordNLP = new StanfordCoreNLP(props);
    }

    // Load the Vosk model from the specified path with custom keyword list
    public void loadModel(String modelPath, String keywordList) throws IOException {
        Model model = new Model(modelPath);
        recognizer = new Recognizer(model, 16000.0f, keywordList);
    }

    // Recognize speech from an audio file and process with NLP
    public void recognizeFromFile(String filePath) throws IOException {
        try {
            File audioFile = new File(filePath);
            AudioInputStream audioInputStream = AudioSystem.getAudioInputStream(audioFile);

            byte[] buffer = new byte[4096];
            int bytesRead;

            while ((bytesRead = audioInputStream.read(buffer)) != -1) {
                recognizer.acceptWaveForm(buffer, bytesRead);
            }

            String jsonResult = recognizer.getFinalResult();
            System.out.println("Final Recognition Result: " + jsonResult);

            String recognizedText = extractRecognizedText(jsonResult);
            System.out.println("Recognized Text: '" + recognizedText + "'");

            // Process the recognized text with NLP
            processTextWithNLP(recognizedText);

            // Check for the exit command
            if (recognizedText.equalsIgnoreCase("please exit")) {
                // Print the recognized values of the current session to the console
                System.out.println("Recognized values for this session: " + String.join(" ", sessionRecognizedValues));

                // Ask the user for the file name
                System.out.println("Please enter the file name to save the recognized values: ");
                String fileName = scanner.nextLine(); // Ask for the file name

                // Append recognized values of the current session to the specified .txt file
                writeValuesToFile(sessionRecognizedValues, fileName);

                // Open the file in Notepad (optional)
                openNotepad(fileName);

                sessionRecognizedValues.clear(); // Clear session values after writing
                System.out.println("Session ended. Continuing to record...");
                return; // Exit the current method
            } else {
                // Fetch value from the database based on recognized text
                Optional<programodel> commandValue = repository.findByCommand(recognizedText.toLowerCase().trim());
                if (commandValue.isPresent()) {
                    // Add the recognized value to the session list
                    sessionRecognizedValues.add(commandValue.get().getValue());
                    recognizedValues.add(commandValue.get().getValue()); // Add to all recognized values
                    System.out.println("Recognized Value: " + commandValue.get().getValue());
                } else {
                    System.out.println("Command not found in database.");
                }
            }

            audioInputStream.close();

        } catch (UnsupportedAudioFileException e) {
            System.err.println("Unsupported audio format. Please ensure the audio is in 16kHz, 16-bit PCM WAV format.");
            e.printStackTrace();
        }
    }

    // Process the recognized text with NLP
    private void processTextWithNLP(String text) {
        // Create a CoreDocument with the text
        CoreDocument document = new CoreDocument(text);

        // Annotate the document with the NLP pipeline
        stanfordNLP.annotate(document);

        // Print tokens, part-of-speech tags, and named entities for analysis
        for (CoreLabel token : document.tokens()) {
            String word = token.word();
            String pos = token.get(CoreAnnotations.PartOfSpeechAnnotation.class);
            String ner = token.get(CoreAnnotations.NamedEntityTagAnnotation.class);

            System.out.println("Token: " + word + ", POS: " + pos + ", NER: " + ner);
        }
    }

    // Extract recognized text from JSON result
    private String extractRecognizedText(String jsonResult) {
        return jsonResult.replaceAll("^\\{\\s*\"text\"\\s*:\\s*\"([^\"]+)\"\\s*\\}$", "$1");
    }

    // Method to append recognized values to a .txt file
    private void writeValuesToFile(List<String> values, String fileName) {
        try (FileWriter fileWriter = new FileWriter(fileName, true)) { // Append mode
            for (String value : values) {
                fileWriter.write(value + " "); // Write each value followed by a space
            }
            fileWriter.write(System.lineSeparator()); // Add a newline after all values
            System.out.println("Appended recognized values to " + fileName); // Debug print
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    // Method to open a file in Notepad (optional)
    private void openNotepad(String fileName) {
        try {
            Runtime.getRuntime().exec("notepad.exe " + fileName); // Open the specified file in Notepad
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    // Getter for recognized values
    public List<String> getRecognizedValues() {
        return recognizedValues; // Return all recognized values
    }

    // Getter for session recognized values
    public List<String> getSessionRecognizedValues() {
        return sessionRecognizedValues; // Return the current session's recognized values
    }
}

package com.example.demo.service;

import javax.sound.sampled.*;
import java.io.File;
import java.io.IOException;

public class AudioRecorder {
    private AudioFormat audioFormat;
    private TargetDataLine targetLine;

    public void recordAudio(String filePath, int durationInSeconds) {
        audioFormat = new AudioFormat(16000, 16, 1, true, true); // PCM format
        DataLine.Info info = new DataLine.Info(TargetDataLine.class, audioFormat);

        try {
            targetLine = (TargetDataLine) AudioSystem.getLine(info);
            targetLine.open(audioFormat);
            targetLine.start();

            System.out.println("Recording...");

            AudioInputStream audioInputStream = new AudioInputStream(targetLine);
            File audioFile = new File(filePath);
            // Start recording
            new Thread(() -> {
                try {
                    Thread.sleep(durationInSeconds * 1000);
                } catch (InterruptedException e) {
                    e.printStackTrace();
                }
                targetLine.stop();
                targetLine.close();
                System.out.println("Recording finished.");
            }).start();

            // Write audio data to file
            AudioSystem.write(audioInputStream, AudioFileFormat.Type.WAVE, audioFile);
        } catch (LineUnavailableException e) {
            e.printStackTrace();
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
